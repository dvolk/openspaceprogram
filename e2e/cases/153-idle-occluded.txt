NAME idle-occluded
ARGS --frame-cap 60 --force-occluded 1000 --timeout 3 --perf
EXPECT idle: 100 ms/frame while the window is occluded/minimized
EXPECT [idle] window hidden: 100 ms/frame budget (sim ceiling 100 Hz at 10 ticks/frame)
# The frame rate drops to the idle interval...
CHECK 15 <= int(re.search(r"idled\s+(\d+) of those frames", out).group(1)) <= 25
CHECK int(re.search(r"perf summary \((\d+) frames", out).group(1)) < 120
# ...and the sim does NOT: 3 s at 60 Hz is ~180 ticks whether or not the
# window is visible. This is the whole point of throttling instead of
# skipping the draw.
CHECK 170 <= int(re.search(r"logic (\d+) ticks", out).group(1)) <= 200
