# WHO flies, said by the player instead of by the front of the FIFO line. The
# seat window (right-click a capsule in the VAB) is the feature; --vab-assign is
# its headless stand-in, and it opens the same window on the way. 'Bram' goes up
# because the window said so -- and 'Ace', first in line, stays on the pad:
# naming any seat turns auto-fill off entirely.
NAME vab-seat-assign
ARGS --vab res/ships/tanker.json --kerbals e2e/fixtures/three_kerbals.json
ARGS --time-accel 1 --timeout 12 --frame-cap 30
ARGS --vab-assign 3000,0,0,Bram --vab-launch 5000
EXPECT [vab] seat window: 'capsule_1' (1 seats)
EXPECT [vab] assign hook: 'capsule_1' seat 0 = 'Bram'
EXPECT Crew: 'Bram' aboard 'tanker' part 0 (1/1)
EXPECT [vab] launched 'tanker' (7 parts)
FORBID Crew: 'Ace'
FORBID no kerbal left
FORBID Assertion
FORBID Segmentation
FORBID GL_
FORBID error:
LIMIT 90
