ARGS --timeout 14 --time-accel 100 --system res/systems/lag_playground.json --atlas-dump 4000

# Bakes first, because every engine cross-check below is all() over surfmap, which an empty list
# answers TRUE -- a renamed keybinding would delete this case's point while leaving its prose pins
# green (the same trap science/830 documents).
CHECK len(surfmap) == 2
CHECK all(r["body"] == "AltoFrio" and r["map"] == "temperature" and abs(r["lag_tau"] - 1.0) < 1e-9 for r in surfmap)
# THE ROW, at the digit. 145.4 to 314.1 K is the sea-level pair above it; 2.5 km is authoredPeak(),
# the one height the atlas names everywhere -- NOT the grid's tallest sampled ground, which is
# 2201 m, and CHECK 6 pays that difference rather than hiding it.
CHECK re.search(r"ground at peak\s+129\.2 to 297\.8 K at 2\.5 km \(equinox, after its lag\)", re.search(r"\[atlas\] AltoFrio weight=.*?(?=\[atlas\] \w+ weight=|\Z)", out, re.S).group(0))
# CONTAINMENT: no number on this row may live outside the picture. Both bakes of the season are
# asked to hold it, because the grid's extremes move with the declination (99.19/101.06 K cold,
# 316.53/315.93 K hot) and the old row missed on both ends at every instant.
CHECK (lambda pk: all(pk[0] > r["temp"][0] + 10.0 and pk[1] < r["temp"][2] - 10.0 for r in surfmap))((lambda m: (float(m.group(1)), float(m.group(2))))(re.search(r"\[atlas\] AltoFrio weight=.*?ground at peak\s+([\d.]+) to ([\d.]+) K", out, re.S)))
# ONE HEIGHT TERM, star-independent: the summit pair is the sea-level pair shifted, so the two
# ends move together and the span is what the equator wears, unchanged.
CHECK (lambda eq, pk: eq[0] - pk[0] > 0.0 and eq[1] - pk[1] > 0.0 and abs((eq[0] - pk[0]) - (eq[1] - pk[1])) < 0.2 and abs((eq[1] - eq[0]) - (pk[1] - pk[0])) < 0.2)((lambda m: (float(m.group(1)), float(m.group(2))))(re.search(r"\[atlas\] AltoFrio weight=.*?ground at equator\s+([\d.]+) to ([\d.]+) K", out, re.S)), (lambda m: (float(m.group(1)), float(m.group(2))))(re.search(r"\[atlas\] AltoFrio weight=.*?ground at peak\s+([\d.]+) to ([\d.]+) K", out, re.S)))
# ... AND THAT SHIFT IS THE FILE'S OWN LAPSE at the printed peak: relief=[floor, ceiling, K at the
# floor, K at the ceiling] is the profile measured on the grid (-0.0065 K/m x 2201 m = -14.31 K),
# and the dossier pays it at authoredPeak() rather than at the grid's tallest sampled ground, so
# the ratio is the whole difference: 145.4 - 129.2 = 16.2 against -14.31 x 2500/2201 = -16.25.
# Tolerance 1.0 K covers the half-decimal the row prints and the printed "2.5 km" standing for a
# height rounded to two significant digits.
CHECK (lambda sh, hm: all(abs(sh + r["relief"][3] * (float(hm.group(1)) * (1000.0 if hm.group(2) == "km" else 1.0)) / r["relief"][1]) < 1.0 for r in surfmap))(float(re.search(r"\[atlas\] AltoFrio weight=.*?ground at equator\s+([\d.]+)", out, re.S).group(1)) - float(re.search(r"\[atlas\] AltoFrio weight=.*?ground at peak\s+([\d.]+)", out, re.S).group(1)), re.search(r"\[atlas\] AltoFrio weight=.*?ground at peak\s+[\d.]+ to [\d.]+ K at ([\d.]+) (km|m)", out, re.S))
# THE PROSE BELONGS TO THE KNOB: a summit row says "after its lag" exactly where its own block
# prices an ACTIVE lag -- a dashed one, or a stated 0.0 that leaves the law's ends alone, must not
# claim a lag it does not have. Honest about its reach: in THIS dump exactly one block prints a
# summit row at all (AltoFrio), so the leg's negative side has no subject here and the leg only
# pins the positive half; the unlagged half is science/724's job, whose Earth needle now ends at
# `6.5 km` so a stray parenthetical on that row reddens it too.
CHECK all((("after its lag" in b) == ((lambda m: m is not None and float(m.group(1)) > 0.0)(re.search(r"ground lag\s+([-\d.]+) rad", b)))) for b in re.split(r"\n(?=\[atlas\] \w+ weight=)", out) if "ground at peak" in b)
