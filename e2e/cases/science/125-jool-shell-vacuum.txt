NAME jool-shell-vacuum
CLAIM A Jool-shell experiment classifies the vacuum situation.
KIND golden
ARGS --time-accel 1 --timeout 3 --drag-log --orbit-interval 0.25
ARGS --startship racer,res/ships/racer.json,Jool,inertial-orbit
EXPECT [drag]
# #552 A3: outside the air there is no air to report, so the four fields the game derives
# from a scale height (T/p/a/Mach) DASH rather than quoting 0 K of vacuum. Pinned twice on
# purpose: EXPECT asks for the literal tail on the line, because the parser's air group is
# OPTIONAL -- a CHECK on key absence alone also passes if the game prints no tail at all,
# or prints it in a shape the parser cannot read. The CHECK then says what that tail means
# to a case: no keys, not keys holding zero.
EXPECT T=- p=- a=- Mach=-
CHECK len(drag) >= 2
# Jool's air ends at an authored 275 km since #682 step 3, so its shell is 330 km and an
# inertial bed is 1.25x that = 412.5 km. The old bound was the then-ceiling, 200 km; left
# alone it would now sit 75 km UNDER the air and pass a hull that was scraping it.
CHECK all(d["alt_asl"] > 300000.0 for d in drag)
CHECK all(d["rho"] == 0.0 for d in drag)
CHECK all(d["F"] == 0.0 for d in drag)
CHECK all("airT" not in d and "mach" not in d for d in drag)
FORBID error:
