NAME skiff-trim-and-weathervane
CLAIM The redesigned skiff is the airframe the design point was written for: 626 kg on 21.6 m2 of AR-6.7 wing, one smooth fuselage, holding +3.0..+3.3 deg of angle of attack at 76 m/s and 7 km with 0.39 m of static margin. A one-second rudder kick swings Beta 6 deg and settles; the wing bill is 0.17 m2 of induced drag, not 0.86.
KIND golden
ARGS --time-accel 1 --startship skiff,res/ships/skiff.json,Kerbin,flying-cruise
ARGS --sim-steps 3000 --drag-log --mass-log
ARGS --sim-press 500,3000,R
ARGS --sim-press 500,20000,T
ARGS --sim-press 25000,1000,A
EXPECT [drag]
EXPECT [mass]
CHECK len(drag) >= 40
# Cruise, held: the trim is a wing station and a wing-loading, not a stick.
CHECK all(2.5 < d["aoa"] < 4.0 for d in drag if d["aoa"] is not None)
CHECK min(d["v"] for d in drag if d["t"] <= 20.0) > 60.0
CHECK min(d["alt"] for d in drag if d["t"] <= 20.0) > 6950.0
# Static margin positive and steady (the VAB number, in flight).
CHECK all(0.30 < d["sm"] < 0.50 for d in drag if d.get("sm") is not None)
# The kick: 6 deg of sideslip and back inside 2 deg within 10 s.
CHECK 3.0 < max(abs(d["beta"]) for d in drag if d.get("beta") is not None) < 10.0
CHECK all(abs(d["beta"]) < 2.5 for d in drag if d["t"] >= 40.0)
# The bill: one body at 0.11 x 0.87 m2, wings at 0.010, induced 0.17.
CHECK 0.80 < min(d["a"] for d in drag if d.get("a")) < 1.00
CHECK all(0.35 < d["profile"] < 0.55 for d in drag if d.get("profile") is not None)
CHECK max(d["induced"] for d in drag) < 0.80
CHECK all(-0.55 < d["formcp"] < -0.35 for d in drag if d.get("formcp") is not None)
FORBID error:
