NAME dock-present-while-paused
CLAIM A space dock is positioned beside the ship at spawn even while the game is paused.
KIND invariant
ARGS --time-accel 0 --startship racer,res/ships/racer.json,Kerbin,pad,site=kerbin_dock
ARGS --timeout 4 --dbg-log --orbit-log --orbit-interval 0.5
EXPECT [bldg]
EXPECT [dbg]
EXPECT [orbitlog]
# The sim never advances (time_accel 0), yet the dock must already be driven to
# the spawn time: its collider live and the ship right beside it, at rest in
# the dock's bubble (#850). Before the fix the pose was only driven inside the
# unpaused tick, so the dock was invisible.
CHECK [b for b in bldg if b["id"]=="kerbin_dock"][-1]["live"] == 1
CHECK "bubble" in orbit[-1]["frame"]
# [dbg] is bubble-local: the spawn offset is 50 m along the dock's -Z.
CHECK 100 < sum(p*p for p in dbg[-1]["pos"]) < 10000
CHECK abs(dbg[-1]["v"]) < 0.5
FORBID error:
