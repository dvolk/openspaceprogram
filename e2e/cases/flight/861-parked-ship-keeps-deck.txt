NAME parked-ship-keeps-deck
CLAIM Switching away parks a deck ship on rails in the platform's frame; switching back wakes it onto the deck.
KIND invariant
# The FIRST startship is the active one: a racer on Earth's default pad. The
# skiff on the Venus platform parks FROZEN on rails in the platform's bubble
# (#850) -- the deck gates out because nothing live needs it: the parked ship
# rides the building's frame, not the collider.
ARGS --system res/systems/solar_system.json --time-accel 1
ARGS --startship ferry,res/ships/racer.json,Earth,pad
ARGS --startship skiff,res/ships/skiff_gear.json,Venus,pad,site=venus_cloud_platform
ARGS --timeout 16 --dbg-log --orbit-log --orbit-interval 1.0
ARGS --sim-press 9000,0,F6
EXPECT [bldg]
EXPECT [dbg]
EXPECT [orbitlog]
# While the player flies the Earth ship, the platform's gate is CLOSED.
CHECK any(b["live"] == 0 for b in bldg if b["id"]=="venus_cloud_platform")
# F6 hands control to the skiff: it wakes in place and the deck gates back in.
CHECK [b for b in bldg if b["id"]=="venus_cloud_platform"][-1]["live"] == 1
# Still standing on the deck, at rest in the bubble frame.
CHECK "bubble" in orbit[-1]["frame"]
CHECK abs(dbg[-1]["v"]) < 1.0
CHECK sum(p*p for p in dbg[-1]["pos"]) < 10000
