NAME platform-spawn-rests
CLAIM A ship spawned on the Venus balloon platform rests on the deck, at rest in the platform's frame.
KIND invariant
ARGS --system res/systems/solar_system.json --time-accel 1
ARGS --startship skiff,res/ships/skiff_gear.json,Venus,pad,site=venus_cloud_platform
ARGS --timeout 20 --dbg-log --orbit-log --orbit-interval 1.0
EXPECT [dbg]
EXPECT [bldg]
EXPECT [orbitlog]
# The deck's collider is gated in beside the ship, and the platform holds its
# float radius (sea level 6051800 + 56000).
CHECK [b for b in bldg if b["id"]=="venus_cloud_platform"][-1]["live"] == 1
CHECK all(abs(b["r"] - 6107800) < 1 for b in bldg if b["id"]=="venus_cloud_platform")
# The ship rides the platform's SoI bubble (#850): its frame IS the building,
# so it is at rest like a ship on a pad -- no wind velocity to inherit, no
# deck carry. The bubble frame is named after the building.
CHECK "bubble" in orbit[-1]["frame"]
# At rest relative to the deck: [dbg] vel is frame-local, and in the bubble
# the deck stands still. A ship scrubbed off the deck would read m/s here.
CHECK abs(dbg[-1]["v"]) < 1.0
# Rigid rest: the gear-spawned skiff stands stable on its wheels -- the spread
# of its deck-local offset stays under a metre (a tipping hull moves ~2 m).
CHECK (lambda ps: max(ps) - min(ps) < 1.0)([sum(p*p for p in d["pos"])**0.5 for d in dbg])
# On the deck: [dbg] pos is bubble-local (deck origin at 0), so a bounded
# radius IS "still on the deck" -- no cross-frame distance arithmetic.
CHECK sum(p*p for p in dbg[-1]["pos"]) < 10000
# The body-centred radius holds the float height in ANY frame ([orbitlog] r):
# the ship never falls, from the first log row to the last. (Not [dbg] alt:
# that is AGL, and Venus's relief under the platform is ~120 m below datum.)
CHECK all(abs(o["r"] - 6107800) < 100 for o in orbit)
FORBID error:
