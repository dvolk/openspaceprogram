NAME jet-flight-envelope
CLAIM The redesigned jet (626 kg-class fuselage instead of a barrel stack) pays a 0.87 m2 body silhouette plus a 0.30 m2 surface profile at cruise, and flies at 176 m/s with L/D ~10 -- where the old barrel hull read ~1. A 3 s full-stick pull from trim still throws it into the deep stall (15 deg) before the tail and the hull bring it back inside 6 deg.
KIND golden
ARGS --time-accel 1 --startship jet,res/ships/jet.json,Kerbin,flying-slow
ARGS --sim-steps 840 --drag-log --mass-log
ARGS --sim-press 1000,3000,R
ARGS --sim-press 1500,6000,T
ARGS --sim-press 3000,3000,W
EXPECT [drag]
EXPECT [ctrl]
CHECK len(drag) >= 10
# Wings-level and trimmed in the first rows (the elevons carry 0.5 deg of trim).
CHECK all(d.get("aoa") is not None and abs(d["aoa"]) < 4.0 for d in drag[:3])
# The envelope: a held full-stick pull peaks at 15 deg (measured; the old
# barrel-hull jet went to 88) and settles inside 6 deg from t=11.
CHECK 9.1 < max(abs(d["aoa"]) for d in drag if d["aoa"] is not None) < 25.8
CHECK all(abs(d["aoa"]) < 6.0 for d in drag if d["t"] >= 11.0)
# The bill. BODY silhouette is the pod + fins (0.87 m2 at Cd 0.11); the surfaces'
# own profile is 0.30 m2 at cd 0.010. Induced is the system bill (#556): the wing
# pair is one AR-4 wing, the elevons one AR-4 pair.
CHECK 0.20 < min(d["profile"] for d in drag if d.get("profile")) < 0.40
CHECK any(d.get("induced", 0.0) > 4.0 for d in drag)
# One vector, one law: on the calm rows the force printed IS the law printed.
CHECK 0.90 < max(d["L"] / d["F"] for d in drag if d["F"] > 0.0) < 1.30
# Form drag sits behind the CG (the pod's own cp is nearly on it, the fins are aft).
CHECK any(d.get("formcp") is not None for d in drag)
CHECK all(-0.85 < d["formcp"] < -0.2 for d in drag if d.get("formcp") is not None)
# The stick moved: the elevons reach their travel stop.
CHECK any(d["axis"] == "pitch" and abs(d["deg"]) > 15.0 for d in ctrl)
