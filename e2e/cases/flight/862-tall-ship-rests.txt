NAME tall-ship-rests
CLAIM A tall gearless rocket stands vertical on the platform and rests rigidly, not tips or slides.
KIND invariant
ARGS --system res/systems/solar_system.json --time-accel 1
ARGS --startship heavy,res/ships/heavy_one.json,Venus,pad,site=venus_cloud_platform
ARGS --timeout 25 --dbg-log --orbit-log --orbit-interval 1.0
EXPECT [dbg]
EXPECT [bldg]
EXPECT [orbitlog]
# In the platform's bubble (#850) the deck is static and the tall hull rests
# on it like on a pad -- no carry, no wind to ride: frame-local speed ~0.
CHECK "bubble" in orbit[-1]["frame"]
CHECK abs(dbg[-1]["v"]) < 1.0
# Rigid stand: [dbg] pos is bubble-local (deck origin at 0), so its spread
# across the run IS the tip/slide measure -- a tipping rocket's COM height
# drops, a sliding one's offset grows. Under a metre over the whole run.
CHECK (lambda ps: max(ps) - min(ps) < 1.0)([sum(p*p for p in d["pos"])**0.5 for d in dbg])
# Still at the float height, first row to last (body-centred radius,
# frame-independent; [dbg] alt is AGL and Venus's relief is below datum here).
CHECK all(abs(o["r"] - 6107800) < 100 for o in orbit)
FORBID error:
