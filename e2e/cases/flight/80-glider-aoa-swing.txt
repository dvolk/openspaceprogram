NAME glider-aoa-swing
CLAIM A glider's AoA swings toward the trimmed angle and settles.
KIND invariant
ARGS --time-accel 1 --startship glider,res/ships/glider.json,Duna,pad
ARGS --timeout 10 --drag-log
ARGS --sim-press 1000,2000,R
ARGS --sim-press 1500,3000,T
ARGS --sim-press 5500,2500,W
EXPECT [drag]
CHECK len(drag) >= 2
CHECK any(d["F"] > 0 and d["rho"] > 0 for d in drag)
CHECK max(abs(d.get("aoa", 0.0)) for d in drag) > 30
CHECK max(d.get("a", 0.0) for d in drag) > min(d.get("a", 0.0) for d in drag) * 1.05
# The area-weighted Cd mean moves as the airframe turns -- but NOT monotonically,
# which is worth pinning as loudly as the fact that it moves at all. Nose-on it is
# 0.78; it DIPS to 0.66 near |AoA| 26 deg, where the wings (cd 0.2 nose-on) take
# over the area weighting as their planforms come into the wind, then climbs to
# 0.82 by 58 deg with the pod and fins leading. This check used to compare the
# widest-AoA row against the narrowest and require a 5% rise: with #463's mount
# fix the far end of the swing reaches 4.7%, and the row it compared against was a
# pad row whose flow direction is meaningless (|F| = 0). So pin the spread over the
# rows where air actually pushed: the Cd in [drag] is a property of the PART MIX
# facing the flow, not of the shape alone.
CHECK (lambda ds: (ds and max(d["cd"] for d in ds) > min(d["cd"] for d in ds) * 1.05))([d for d in drag if d.get("aoa") is not None and d.get("cd") is not None and d["F"] > 0.0])
