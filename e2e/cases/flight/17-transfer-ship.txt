NAME transfer-ship
CLAIM A near-co-orbital rendezvous is a nudge, not a launch-class burn.
KIND invariant
ARGS --startships e2e/fixtures/startships.json --time-accel 1 --timeout 12
ARGS --xfer-log --orbit-interval 0.5 --transfer-target transporter
ARGS --sim-press 1500,50,F6
ARGS --sim-press 2500,50,F6
ARGS --sim-press 3500,50,F6
EXPECT [xferlog]
CHECK len(xfer) >= 2
CHECK last(xfer)["target"] == "transporter"
CHECK last(xfer)["dv_dep"] > 0 and last(xfer)["dv_dep"] < 20000
CHECK last(xfer)["dv_cap"] == 0
# A ship target has no capture burn, so the capture pair is ABSENT rather than
# a 0 that would read as a burn at the target's centre (#405).
CHECK last(xfer)["r_cap"] is None
CHECK last(xfer)["r_cap_asl"] is None
CHECK last(xfer)["total"] == last(xfer)["dv_dep"]
CHECK last(xfer)["tof"] > 0
# A near-co-orbital rendezvous is a nudge, not a launch: if the phasing
# families (#136) drop out of the search this jumps by an order of magnitude.
# Which family wins depends on the ToF sample grid, so it is not pinned here;
# the multi-revolution legs themselves are pinned in tests/test_transfer.cpp.
CHECK last(xfer)["dv_dep"] < 1000
FORBID error:
