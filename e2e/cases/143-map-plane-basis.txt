# The orbit map's plane basis, per combo slot (#181). The map is pure drawing,
# so a flipped e2 is invisible to every check but an eyeball: --map-dump prints
# what each slot actually draws with, from the same mapPlaneBasis the two map
# windows call.
#
# Kerbin on a POLAR bed, so the three slots take three different paths through
# OrbitMap::setPlane: Equatorial pins the basis to the focus frame's +X,
# Ecliptic takes the canonical X/Z one, Orbital falls back to the node line --
# its h lies 20 deg from the focus +X, inside the cone where +X cannot define a
# stable east (#185 left that fallback in place, and case 147 pins the pinned
# path). All three must agree on handedness -- a prograde body at +e1 moves
# along -e2, printed as sweep=-1.
NAME map-plane-basis
ARGS --time-accel 1 --timeout 6 --map-dump 2500 --startship racer,res/ships/racer.json,Kerbin,high-polar
FORBID GL_
FORBID error:
CHECK len(mapb) == 3
CHECK [r["slot"] for r in mapb] == ["equ", "ecl", "orb"]
CHECK all(r["sweep"] == -1 for r in mapb)
# Each slot's (e1, e2) is an orthonormal pair spanning the plane it claims.
CHECK all(abs(sum(a*a for a in r["e1"]) - 1.0) < 1e-5 for r in mapb)
CHECK all(abs(sum(a*a for a in r["e2"]) - 1.0) < 1e-5 for r in mapb)
CHECK all(abs(sum(a*b for a,b in zip(r["e1"], r["e2"]))) < 1e-5 for r in mapb)
CHECK all(abs(sum(a*b for a,b in zip(r["n"], r["e1"]))) < 1e-5 for r in mapb)
CHECK all(abs(sum(a*b for a,b in zip(r["n"], r["e2"]))) < 1e-5 for r in mapb)
# Equatorial means the POLE, not the rail normal (#173): on Kerbin the two
# slot normals sit exactly the 23.44 deg axial tilt apart.
CHECK abs(sum(a*b for a,b in zip(mapb[0]["n"], mapb[1]["n"])) - 0.917477) < 1e-5
# And Orbital means the SHIP'S plane: a polar bed puts h across the pole.
CHECK abs(sum(a*b for a,b in zip(mapb[2]["n"], mapb[0]["n"]))) < 1e-5
# That h also lies inside the +X degeneracy cone (20 deg from it, tolerance 26),
# so this slot really is on the node-line fallback rather than the pinned path
# case 147 covers. The bed puts h along the Equatorial slot's own screen-x, so
# the fallback basis is perpendicular to it -- pinned, so widening the 0.44
# tolerance stays a decision instead of silently moving this view.
CHECK abs(sum(a*b for a,b in zip(mapb[2]["n"], mapb[0]["e1"])) - 1.0) < 1e-5
CHECK abs(sum(a*b for a,b in zip(mapb[2]["e1"], mapb[0]["e1"]))) < 1e-5
