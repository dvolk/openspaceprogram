# pod materials study -> EVA take/store -> recover scores the sample
NAME materials-pod-dance
ARGS --time-accel 1 --timeout 9 --width 1920 --height 1080
ARGS --startship labpod,res/ships/labpod.json,Kerbin,rot-orbit
ARGS --ui-focus 1000,labpod/materials_pod_1
ARGS --sim-mouse 1100,0,CUR,CUR,RMB
ARGS --ui-click 1500,"labpod > Materials Pod (2m)##1/Run Experiment##pod_run"
ARGS --ui-focus 2000,labpod/capsule_1
ARGS --sim-mouse 2100,0,CUR,CUR,RMB
ARGS --ui-click 2500,"labpod > Capsule (2m)##0/EVA##eva0"
ARGS --ui-click 3000,"labpod > Materials Pod (2m)##1/Take  (to Ash)##pod_take"
ARGS --ui-click 3500,"labpod > Capsule (2m)##0/Store  (into capsule)##store_Ash"
ARGS --ui-click 4000,"labpod > Capsule (2m)##0/Board##board_Ash"
ARGS --recover-anywhere
ARGS --sim-press 4900,0,ESCAPE
ARGS --ui-click 5000,"Game Menu/Recover Vessel"
# Every step is a click a player could make: RMB-pick a part to open its window
# (--ui-focus aims the cursor at the part, so the case names parts, never
# pixels), then press the window's button. The part windows stay open once
# opened, so only the two RMB picks need aiming; the four button clicks address
# windows by name.
# The Take label names the courier and the Store/Board paths carry a hidden id that
# names it too ("Take  (to Ash)", "Store  (into capsule)##store_Ash",
# "Board##board_Ash"), so the paths pin who the game thinks the courier is -- a wrong
# courier fails the click outright instead of moving the sample somewhere else.
# The walker boards before the flight can end: a free kerbal is not a mission
# (recoverActive refuses), so the crew is back in the capsule before the click.
EXPECT [science] t=
EXPECT recorded
EXPECT materials study
EXPECT moved
EXPECT [science] recovered +31
EXPECT [recover] t=
EXPECT [scene] spacecenter -> spacecenter (enterSpaceCenter)
EXPECT [uifocus] 'labpod/materials_pod_1' ->
EXPECT ship=labpod part=1 (materials_pod)
EXPECT ship=labpod part=0 (capsule)
EXPECT EVA: 'Ash' out of 'labpod' part 0
EXPECT Board: 'Ash' into 'labpod' part 0
EXPECT moved 'Low orbit materials study on Kerbin' Materials Pod (2m) -> Kerbal
EXPECT moved 'Low orbit materials study on Kerbin' Kerbal -> Capsule (2m)
