# Quit to title from the VAB. The player path: open the editor from the hub
# (so the hub is the frame below), "Back to game" / ESC to return to it, then
# "Return to title". --quit-title used to call resetToTitle from inside the
# VAB, a transition nothing offers -- and with --vab-empty the floor under the
# editor is the title itself, so there is no hub to walk through at all.
NAME vab-quit-title
ARGS --time-accel 1 --timeout 10 --startship racer,res/ships/racer.json,Kerbin,rot-orbit
ARGS --sim-press 1000,0,ESCAPE --ui-click 1500,"Game Menu/VAB"
ARGS --sim-press 2500,0,ESCAPE
ARGS --ui-click 3000,"Game Menu/Return to title" --ui-click 3500,"Question/Return"
EXPECT [scene] flight -> spacecenter (push)
EXPECT [scene] spacecenter -> vab (push)
EXPECT [scene] vab -> spacecenter (pop)
EXPECT [game] unloaded: ships torn down
EXPECT [scene] spacecenter -> title (enterTitle)
