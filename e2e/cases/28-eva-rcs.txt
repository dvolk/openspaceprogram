NAME eva-rcs
ARGS --time-accel 1 --timeout 8 --eva-log --orbit-log --orbit-interval 0.5 --sim-press 2000,2000,W,5000,1500,R --frame-cap 30 --startship kerbal,res/ships/kerbal.json,Kerbin,inertial-orbit
EXPECT Building ship 'kerbal' (1 parts)
FORBID GL_
FORBID error:
CHECK len(eva) > 5
CHECK all(e["mode"] == "space" for e in eva)
CHECK abs(last(eva)["vel"][0] - first(eva)["vel"][0]) > 2.0
CHECK abs(last(orbit)["E"] - first(orbit)["E"]) > 100.0
# #242: rcsFiring is what the render pass gates the RCS puff on. Kerbal armed
# its own shadowed rcsDir and never set the flag, so EVA thrust drew nothing.
CHECK any(e["rcs"] == 1 for e in eva)
# And it must not stick on once the key is up (cleared per tick by clearRcs):
# the W press starts at 2000ms wall, so t<1.5 is before any thrust.
CHECK all(e["rcs"] == 0 for e in eva if e["t"] < 1.5)
