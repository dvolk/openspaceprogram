# every scene entry restores visibility (Game::ensure_ui_visible), so the
# hub cannot come up as a blue screen with its chrome hidden. The log line
# is the anchor (it fires only when the hide was actually live). The hide is
# re-armed before each of the three transitions so the push AND pop paths
# both run the reset branch.

ARGS --ship res/ships/racer.json --scenario rot-orbit --time-accel 1 --timeout 10
ARGS --sim-press 1000,0,TAB --space-center 2000 --sim-press 2500,0,TAB
ARGS --tracking 3000 --sim-press 3500,0,TAB --tracking-close 4000
EXPECT [scene] flight -> spacecenter (push)
EXPECT [ui] TAB-hide dropped at scene entry
EXPECT [scene] spacecenter -> tracking (push)
EXPECT [scene] tracking -> spacecenter (pop)
EXPECT Timeout reached (10.
