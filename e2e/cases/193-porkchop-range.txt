NAME porkchop-range
ARGS --system res/systems/solar_system.json --time-accel 1 --timeout 15
ARGS --startship racer,res/ships/racer.json,Sun,heli-earth --transfer-target Mars
ARGS --porkchop-log --porkchop-n 24 --orbit-interval 0.5
# The departure/ToF range sliders topped out at 604800 s (7 days) and clamped
# the range they seeded, so switching them on turned a Mars sweep into a
# one-week sweep without saying so. They now scale with the target's own
# period, which is also what puts a 1-revolution phasing window in reach (#136).
ARGS --ui-click 2000,"Windows/Transfer"
ARGS --ui-click 2600,"Transfer/Porkchop"
ARGS --sim-press 3000,0,P
ARGS --ui-click 4000,"Porkchop/Departure range"
ARGS --ui-click 4300,"Porkchop/ToF range"
ARGS --sim-press 5000,0,P
EXPECT [porkchop]
CHECK len(porkchop) >= 2
# Auto range for Mars: depart 0 .. one period (5.9e7 s), ToF 60 s .. three.
CHECK last(porkchop)["dep_hi"] > 5.0e7
CHECK last(porkchop)["tof_hi"] > 1.0e8
# Enabling the sliders must not shrink the window the sweep searches.
CHECK porkchop[0]["tof_hi"] == last(porkchop)["tof_hi"]
CHECK porkchop[0]["dep_hi"] == last(porkchop)["dep_hi"]
CHECK last(porkchop)["tof_lo"] <= 60 and last(porkchop)["dep_lo"] == 0
CHECK last(porkchop)["tof_min"] <= last(porkchop)["tof_hi"]
CHECK last(porkchop)["t_dep_min"] <= last(porkchop)["dep_hi"]
FORBID GL_
FORBID error:
