ARGS --sim-mouse 1100,0,CUR,CUR,RMB

ARGS --ui-click 3000,"racer > Capsule (2m)##0/Board##board_Ash"
# --ui-focus aims the cursor at the capsule by PART, so the case names no pixel:
# the RMB pick that opens the part window used to be a hardcoded 960,540, which
# only meant "the middle of the screen, where the default camera happens to
# frame the capsule". The window path is qualified now that a part window's name
# is stable ("racer > Capsule (2m)##0"); it used to carry the Vehicle pointer,
# which left a bare "Board" as the only writable form -- and a bare label is
# satisfied by ANY window's Board, so the case could have pressed a different
# capsule's button and still passed.
# Boarding deletes the walking Kerbal and parks its suit in the capsule as a seat
# (part.h) -- and the loop that drew the button kept the pointer and read
# k->parts immediately after: heap-use-after-free at gameui.cpp:2198 under ASan.
# A release run cannot see that read itself; the proof is these same args under
# osp_asan. What this case owns is keeping the Board button exercised at all --
# until now the only board coverage was the --board hook, which never touches UI.
EXPECT [pick] t=
EXPECT EVA: 'Ash' out of 'racer' part 0
EXPECT Board: 'Ash' into 'racer' part 0
EXPECT Active ship 1 of 1: racer
