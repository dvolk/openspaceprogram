# two findings couriered home; only the non-pod one scores as new
NAME courier-two-findings
ARGS --time-accel 1 --timeout 10 --startship labpod,res/ships/labpod.json,Kerbin,rot-orbit
ARGS --experiment 1000
ARGS --pod-experiment 1500
ARGS --eva 2000
ARGS --take 2500
ARGS --store 3000
# Switch control back to the ship (F6) rather than boarding: boarding would carry
# the suit's still-held study home, and the point here is that a finding left on
# a walking crew member is NOT banked. "Recover Vessel" needs a ship active, so
# this switch is what a player would do.
ARGS --sim-press 3500,50,F6
ARGS --space-center 4000 --recover 5000 --recover-anywhere
EXPECT recorded 'Low orbit observation
EXPECT recorded 'Low orbit materials study
EXPECT moved 'Low orbit materials study
EXPECT moved 'Low orbit observation
EXPECT new: Low orbit observation
FORBID new: Low orbit materials study
EXPECT [recover] t=
EXPECT [spacecenter] entered
FORBID GL_
FORBID error:
