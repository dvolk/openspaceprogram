NAME reload-shipless
CLAIM An in-game reload of a shipless save moves to the hub.

# Center hub, not the title. The bare boot is the title front door; the
# shipless reload moves it to the hub.
SEED e2e/fixtures/save_shipless saves/20260101_000000-reload/save1
ARGS --system res/systems/ksp_system.json --timeout 10
ARGS --ui-click 1500,"Title Menu/Save/Load"
ARGS --ui-click 3000,"Save/Load/Load##saveload"
EXPECT [boot] no vessel: title screen
EXPECT saves/20260101_000000-reload/save1 (active: (none))
EXPECT [scene] title -> spacecenter (enterSpaceCenter)
