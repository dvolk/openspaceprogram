# The EVA/Board buttons live in the capsule's part window, so the case RMB-picks
# the capsule first -- by NAME (--ui-focus resolves the pixel) rather than by a
# centre coordinate: #876's ground-framed pad camera puts the hull's COM at dead
# centre, and `racer`'s COM is in its fuel tank, so the old 960,540 opened the
# tank's window and the EVA row was simply never drawn (#946).
NAME platform-eva-stands
CLAIM EVA on the platform: stand, walk, fall off the edge, and keep flying with RCS.
KIND invariant
ARGS --system res/systems/solar_system.json --time-accel 1
ARGS --startship racer,res/ships/racer.json,Venus,pad,site=venus_cloud_platform
ARGS --timeout 34 --eva-log --orbit-interval 0.5
ARGS --ui-focus 2000,"racer/capsule_1"
ARGS --sim-mouse 2100,0,CUR,CUR,RMB
ARGS --ui-click 3000,"racer > Capsule (2m)##0/EVA##eva0"
ARGS --sim-press 6000,22000,W
EXPECT EVA: 'Ash' out of 'racer' part 0
EXPECT [evalog]
# The walker steps out at the capsule's height (pad semantics), falls to the
# deck and stands: bubble-local z ~ restAlt, ground mode -- not teleported a
# planet away by a body-centric floor read (#850).
CHECK all(e["mode"] == "ground" for e in eva if 5.0 < e["t"] < 19.0)
CHECK all(abs(e["pos"][2]) < 5.0 for e in eva if 5.0 < e["t"] < 19.0)
# Walking works on the deck: horizontal displacement while ground mode holds.
CHECK (lambda es: (es[-1]["pos"][0]-es[0]["pos"][0])**2 + (es[-1]["pos"][1]-es[0]["pos"][1])**2 > 0.25)([e for e in eva if 6.0 < e["t"] < 8.0])
# No bounce limit cycle: the standing height holds still while walking (the
# floor guard and the contact agree on the deck's margin, #850).
CHECK (lambda zs: max(zs) - min(zs) < 0.3)([e["pos"][2] for e in eva if 9.0 < e["t"] < 15.0])
# W keeps holding past the deck edge (~35 m out): the suit falls off...
CHECK any(e["mode"] == "space" and e["pos"][2] < 0.0 for e in eva if e["t"] > 21.0)
# ...and KEEPS RESPONDING: W is RCS in space mode, so the fall shows burns.
# (The floor guard's off-edge branch must not skip the space-mode laws.)
CHECK any(e["rcs"] == 1 for e in eva if e["t"] > 21.0)
