NAME terrain-lod-drill
CLAIM The terrain queue builds collision under the camera early.
KIND invariant
ARGS --startship racer,res/ships/racer.json,Kerbin,pad --width 1920 --height 1080
ARGS --terrain-px 128 --timeout 12 --terrain-log --orbit-interval 1
EXPECT [terrain]
EXPECT Main loop starting
FORBID error:
CHECK len(terrain) > 3
CHECK last(terrain)["deepest"] == last(terrain)["max_depth"] == 12
CHECK last(terrain)["collision"] > 0
CHECK last(terrain)["deep_off"] < 1000.0
# The ground has to arrive EARLY, not eventually. One worker builds the patch
# grids, so the queue's order decides what exists next -- and collision meshes
# are only built at max_depth, so a tree that stays shallow under the camera
# has no ground under the ship. Before the queue was ranked by screen error,
# 20 s at this LOD reached depth 8 with collision=0 for the whole run.
CHECK any(t["collision"] > 0 for t in terrain[:3])
LIMIT 90
