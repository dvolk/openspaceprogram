NAME surfmap
CLAIM The Surface Map draws the body's shaded surface.
KIND invariant
ARGS --time-accel 1 --startship racer,res/ships/racer.json,Kerbin,rot-orbit
ARGS --timeout 15 --surfmap-log --surfmap-n 128
ARGS --sim-press 3000,0,M
EXPECT [surfmap]
CHECK len(surfmap) >= 1
CHECK last(surfmap)["body"] == "Kerbin"
CHECK last(surfmap)["w"] == 128 and last(surfmap)["h"] == 64
CHECK last(surfmap)["shade"] == "on"
CHECK all(0.0 <= c <= 1.0 for c in last(surfmap)["albedo"])
CHECK all(s <= a + 1e-9 for s, a in zip(last(surfmap)["shaded"], last(surfmap)["albedo"]))
CHECK any(s < a for s, a in zip(last(surfmap)["shaded"], last(surfmap)["albedo"]))
# #839/#841: a ground picture colours from the ground's height (its palette band, plus the sea
# test when the map asks for one), so it asks the cache -- and here it asks for the first time in
# the run, so `miss`. A dash is an absent QUESTION, and only two shapes print it now: a banded
# body, whose colour comes from a band table and never reads a height, and a refused field, which
# paints flat grey instead of the ground. Pinned here because this is the run shape that prints
# the first of them.
CHECK last(surfmap)["elev_cache"] == "miss"
