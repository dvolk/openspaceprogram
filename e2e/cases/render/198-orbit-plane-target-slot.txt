# The orbit map's fourth plane slot: "Target orbit" -- the TARGET window's
# target seen as a PLANE about the body you orbit. The target list is bodies
# orbiting that body plus vessels in its SOI, and the planner relativizes the
# target to whatever focus frame it is handed, so the normal is always the
# target's own conic about the map's focus. ONE target then drives the transfer
# numbers, the map orientation, the HUD's Inc and the 3D overlay's second ring.
#
# Minmus is authored 6 deg off Kerbin's rail plane, which gives the slot an exact
# answer from data rather than from the code under test: the dihedral must read
# 6.0000 deg, and the slot's normal must sit cos(6 deg) = 0.994522 from the
# Ecliptic slot's while being visibly off Kerbin's EQUATOR (23.44 deg of
# obliquity) -- Minmus's rail plane, not one of the other three relabelled.
NAME orbit-plane-target-slot
CLAIM The Target orbit slot uses the target's conic about the focus.
KIND invariant
ARGS --time-accel 1 --timeout 9 --startship racer,res/ships/racer.json,Kerbin,inertial-orbit
ARGS --transfer-target Minmus --map-plane 3
ARGS --plane-log --orbit-log --xfer-log --orbit-interval 1 --map-dump 7000
ARGS --sim-press 1500,0,ESCAPE
ARGS --ui-click 2500,"Game Menu/Settings"
ARGS --ui-click 3500,"Settings/Orbital plane"
EXPECT [planelog]
EXPECT [xferlog]
FORBID FAILED
FORBID error:
CHECK len(plane) > 3
# The fourth [mapdump] row exists only because a target is picked.
CHECK len(mapb) == 4
CHECK mapb[3]["slot"] == "tgt"
# One target everywhere: the plane slot and the transfer solve agree on it.
CHECK all(x["target"] == "Minmus" for x in xfer)
CHECK all(p["plane"] == "tgt" for p in plane if p["t"] > 1.0)
# The overlay draws the map's target-plane normal -- the render frame IS the
# inertial one here, so the digits must match -- and its own orbit normal matches
# the Orbital slot. Two code paths (render.cpp, gameui.cpp), one plane each.
CHECK all(abs(p["n_ref"][0] - mapb[3]["n"][0]) < 1e-6 and abs(p["n_ref"][1] - mapb[3]["n"][1]) < 1e-6 and abs(p["n_ref"][2] - mapb[3]["n"][2]) < 1e-6 for p in plane if p["t"] > 1.0)
CHECK all(abs(p["n_orb"][0] - mapb[2]["n"][0]) < 1e-6 and abs(p["n_orb"][1] - mapb[2]["n"][1]) < 1e-6 and abs(p["n_orb"][2] - mapb[2]["n"][2]) < 1e-6 for p in plane if p["t"] > 1.0)
# The authored 6 deg, measured three ways: the planner's state lift into the
# overlay's acos, and the HUD's orbitPlaneAngles in tick.cpp.
CHECK all(abs(p["inc"] - 6.0) < 0.001 for p in plane if p["t"] > 1.0)
CHECK all(abs(o["inc"] - 6.0) < 0.001 for o in orbit if o["t"] > 1.0)
# cos(6 deg) from the Ecliptic slot, and NOT from Kerbin's equator (0.8811).
CHECK abs(sum(a*b for a,b in zip(mapb[3]["n"], mapb[1]["n"])) - 0.994522) < 1e-5
CHECK abs(sum(a*b for a,b in zip(mapb[3]["n"], mapb[0]["n"])) - 0.994522) > 0.05
# A node line exists, so the ticks get drawn: that is where the plane change is
# cheapest, and the whole reason the slot is worth having in 3D.
CHECK all(p["ticks"] == "drawn" for p in plane if p["t"] > 1.0)
# No zero longitude was chosen for this plane, so the readout dashes LAN rather
# than invent one -- while still drawing the node it cannot name. (LPe dashes
# too, though here that is the near-circular guard, not the missing axis.)
CHECK all(o["lan"] == "-" for o in orbit if o["t"] > 1.0)
CHECK all(o["lpe"] == "-" for o in orbit if o["t"] > 1.0)
